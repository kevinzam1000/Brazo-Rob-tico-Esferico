Brazo Robótico Esférico

Proyecto Integrador de Mecatrónica

Descripción del Proyecto

Este proyecto consiste en el diseño, análisis cinemático y construcción de un brazo robótico de configuración esférica controlado mediante un microcontrolador Arduino[cite: 1].

El sistema permite calcular y ejecutar trayectorias exactas para posicionar el brazo en coordenadas específicas, utilizando motores paso a paso, sistemas de transmisión por correa y retroalimentación de posición mediante un encoder magnético absoluto[cite: 1].

Componentes Electrónicos y Hardware

El sistema de control y potencia está integrado por componentes de grado mecatrónico, diseñados para garantizar precisión articular, alimentación constante e interfaz de diagnóstico en tiempo real.

Arduino Mega 2560

Microcontrolador principal encargado del procesamiento numérico de la cinemática inversa, generación de pulsos y lectura de entradas digitales[cite: 1, 3].

2x Motores NEMA 17 (17HS4401S)

Actuadores paso a paso bipolares que proporcionan el torque e inercia necesarios para el posicionamiento de los ejes de rotación[cite: 1, 3].

2x Drivers TB6600

Módulos de potencia ajustables con control de micropasos y aislamiento por optoacopladores para alimentar los motores NEMA 17[cite: 1, 3].

Motor 28BYJ-48

Actuador unipolar compacto utilizado para la tracción y control de extensión de la cremallera del brazo[cite: 1, 3].

Driver ULN2003APG

Módulo con arreglo de transistores Darlington encargado de la conmutación de corriente para las bobinas del motor 28BYJ-48.

Fuente Conmutada 12V 10A

Unidad de alimentación regulada de alta capacidad que entrega energía constante a las etapas de potencia sin caídas de tensión.

Pulsadores de Control Físico

Entradas digitales dedicadas: un botón para iniciar la secuencia de movimiento y otro para el habilitador/liberación manual de motores[cite: 3].

Indicadores LED de Estado

Sistema visual de retroalimentación en tiempo real sobre la condición operacional del robot[cite: 3]:

  • Verde: Sistema en ejecución (motores en movimiento)[cite: 3].
  • Naranja: Sistema en estado de pausa o liberación manual[cite: 3].
  • Rojo: Bloqueo de seguridad activado por riesgo de colisión[cite: 3].

Funcionamiento

El usuario introduce las coordenadas deseadas a través de la interfaz o puerto serie, y el sistema calcula cinemáticamente el número de pasos requerido para cada motor[cite: 1, 3].

Durante la ejecución, la señal de pulsos activa los drivers y el encoder AS5600 permite retroalimentar la posición real para evitar desviaciones o pérdida de pasos[cite: 1, 3].

El brazo robótico fue diseñado mediante modelado CAD 3D (SolidWorks)[cite: 1]. Esta sección incluirá los esquemas de transmisión, soportes estructurales, planos de ensamble y dimensiones del mecanismo esférico[cite: 1].

Programación y Control

El firmware en C++ calcula la cinemática inversa a partir de las coordenadas cartesianas de entrada (X, Y, Z) para determinar los ángulos articulados (th1, th2) y la extensión de la cremallera (e)[cite: 3]. Incluye algoritmos de protección anti-colisión en tiempo real, habilitador manual y aceleración gradual para los motores paso a paso[cite: 3].

Brazo_Esferico_Control.ino 
float x = -130;
float y = 150;
float z = -80;
float th1 = 0;
float th1n = 0;
float th2 = 0;
float th2n = 0;
float h = 0;
float e = 0;
float a = 0;

const int enaPin = 34;
const int dirPin = 36;
const int pulPin = 38;

const int enaPin2 = 35;
const int dirPin2 = 37;
const int pulPin2 = 39;

int ppv = 2048;
float con = 0;
float cona = 0;
int pasos = 0;
int pinan = 0;

int paro = 19;
int pausa = 20;
int giro = 21;
int ini = 8;
int inia;
int inic;
int fre = 9;
int fren;
int frea;
int cuen;

void setup() {
  Serial.begin(9600);

  pinMode(enaPin, OUTPUT);
  pinMode(dirPin, OUTPUT);
  pinMode(pulPin, OUTPUT);

  pinMode(enaPin2, OUTPUT);
  pinMode(dirPin2, OUTPUT);
  pinMode(pulPin2, OUTPUT);

  pinMode(2, OUTPUT);
  pinMode(3, OUTPUT);
  pinMode(4, OUTPUT);
  pinMode(5, OUTPUT);

  pinMode(paro, OUTPUT);
  pinMode(giro, OUTPUT);
  pinMode(inic, INPUT);
  pinMode(fre, INPUT);

  digitalWrite(enaPin, HIGH); 
  digitalWrite(enaPin2, HIGH);
  digitalWrite(paro, LOW);

  a = y - 81.5;

  th1 = atan2(x,(z*-1))* (180 / PI);
  h = sqrt(x*x + z*z);
  th2 = atan2(a,h) * (180 / PI);
  e = ((sqrt(a*a + h*h)) - 50); // Va a ser e-78mm porque es lo menos que puede estar la cremallera, 
  con = (e - 78) / 10;          // e incluso menos con la restriccion del espacio
}

void loop() {
  inic = digitalRead(ini);
  Serial.println(e);
  Serial.println(con);
  fren = digitalRead(fre);

  // Habilitador manual
  if (fren == HIGH && frea == LOW){
    cuen = !cuen;   // Cambia entre 0 y 1
    digitalWrite(enaPin, cuen);
    digitalWrite(enaPin2, cuen);
  }

  if (th2 > 45 && e < 113){
    Serial.println("Motor bloqueado por riesgo de colision");
    digitalWrite(paro, HIGH);
  }
  else if (th2 < (-21.6) || e < 55){
    Serial.println("Motor bloqueado por riesgo de colision");
    digitalWrite(paro, HIGH);
  }
  else {
    Serial.println("Todo good");
  
    // Secuencia de giro
    if (inic == HIGH && inia == LOW){ 
      Serial.println("Ya empezo");
      if(con < 0){
        cona = con * (-1);
        pasos = (ppv * cona) / 12.6;
        horario();
      }
    
      if(con > 0){
        pasos = (ppv * con) / 12.6;
        antihorario();
      }
      delay(500);
      
      if (th1 < 0){
        run();
        digitalWrite(dirPin, LOW);
        th1n = th1 * (-1);
        gira1(th1n);
      }
      if (th1 > 0){
        run();
        digitalWrite(dirPin, HIGH);
        gira1(th1);
      }
      delay(500);

      if (th2 < 0) {
        run();
        digitalWrite(dirPin2, LOW);
        th2n = th2 * (-1);
        gira2(th2n);
      }
      if (th2 > 0) {
        run();
        digitalWrite(dirPin2, HIGH);
        gira2(th2);
      }
      delay(500);
    }
  }

  stop();
  delay(50);
  inia = inic;
  frea = fren;
}

void run(){
  digitalWrite(pausa, LOW);
  digitalWrite(giro, HIGH);
}

void stop(){
  digitalWrite(giro, LOW);
  digitalWrite(pausa, HIGH);
}

void gira1(long pul1) {
  long totalPasos1 = (pul1 / 360.0) * 1600.0;

  for (long paso1 = 0; paso1 < totalPasos1; paso1++) {
    int velocidad1 = 800;

    if (paso1 < 10) {
      velocidad1 = map(paso1, 0, 10, 3000, 800);
    }

    digitalWrite(pulPin, HIGH);
    delayMicroseconds(velocidad1);
    digitalWrite(pulPin, LOW);
    delayMicroseconds(velocidad1);
  }
}

void gira2(long pul2) {
  long totalPasos2 = (pul2 / 360.0) * 1600.0;

  for (long paso2 = 0; paso2 < totalPasos2; paso2++) {
    int velocidad2 = 800;

    if (paso2 < 10) {
      velocidad2 = map(paso2, 0, 10, 3000, 800);
    }

    digitalWrite(pulPin2, HIGH);
    delayMicroseconds(velocidad2);
    digitalWrite(pulPin2, LOW);
    delayMicroseconds(velocidad2);
  }
}

void horario (){
  for (int i = 0; i < pasos; i++){
    int pin = i % 4 + 2; 
    digitalWrite(pin, HIGH);
    digitalWrite(pinan, LOW);
    delayMicroseconds(800);
    Serial.println(pin);
    pinan = pin;
  }
}

void antihorario (){
  for (int i = 0; i < pasos; i++){
    int pin = (i % 4 - 5) * (-1); 
    digitalWrite(pin, HIGH);
    digitalWrite(pinan, LOW);
    delayMicroseconds(800);
    Serial.println(pin);
    pinan = pin;
  }
}
